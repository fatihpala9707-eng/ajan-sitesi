let py = null;
self.onmessage = async (e) => {
  const m = e.data;
  try {
    if (m.cmd === 'init') {
      importScripts(m.base + 'pyodide.js');
      py = await loadPyodide({ indexURL: m.base });
      py.runPython(m.harness);
      self.postMessage({ id: m.id, ok: true });
      return;
    }
    py.globals.set('_REQ', JSON.stringify(m));
    const out = py.runPython('_dispatch(_REQ)');
    self.postMessage({ id: m.id, ok: true, data: out });
  } catch (err) {
    self.postMessage({ id: m.id, ok: false, error: String((err && err.message) || err) });
  }
};

${x[1]} kendi __init__'ini yazmış ama miras aldığı sınıfın __init__ metodunu çağırmamış olabilir. Alt sınıf __init__ yazınca üst sınıfınki kendiliğinden çalışmaz.

Miras aldığın sınıfın metodunu çağırırken self'i sen vermezsin; Python onu kendisi ekler. Değilse nesneyi üretirken verdiğin değerlerin sayısı __init__ parametreleriyle tutmuyor.
